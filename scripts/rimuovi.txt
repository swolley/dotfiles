# Packages to REMOVE, not to install: read by the "Cleanup" phase of
# setup-machine.sh, which removes the ones present with pacman -Rns.
#
# EndeavourOS ships a full GNOME session and a handful of its own tools.
# They are not needed under niri + noctalia, but they come with the
# installation image anyway: without this list a fresh machine would end
# up with all of them.
#
# Dependencies pulled in by these packages should not be listed: -Rns
# removes them on its own if nothing else uses them.

# --- GNOME: the minimal fallback session stays (gnome-shell, gdm and the apps
# you actually use), the surrounding extras go
alacarte
dconf-editor
epiphany
evince
extension-manager
gnome-browser-connector
gnome-clocks
gnome-console
gnome-control-center
gnome-menus
gnome-nettool
gnome-power-manager
gnome-screenshot
# its Nautilus entry "Open in Terminal" duplicates ours (kitty + tmux)
gnome-terminal
gnome-tweaks
# extensions only live inside gnome-shell: they do not run under niri
gnome-shell-extensions
gnome-shell-extension-arch-update
gnome-shell-extension-dash-to-dock
gnome-shell-extension-gsconnect

# --- EndeavourOS: welcome app, guides and tools for fresh installs
welcome
eos-quickstart
eos-apps-info
eos-packagelist
eos-log-tool
reflector-simple
# forces purple on the login screen and the GNOME desktop
eos-settings-gnome
# unused GTK theme: colors are generated by noctalia's templates
arc-gtk-theme-eos

# --- Xorg: the session is Wayland. X11 apps run through xorg-xwayland, which
# is a separate package and stays installed
xorg-server
xorg-xinit
xorg-xdpyinfo
xorg-xinput
xorg-xkill

# --- duplicates and replaced tools
# the terminal is kitty
xterm
# file comparison is done with git difftool and lazygit
meld
# btrfs snapshots are managed with snapper from the command line
btrfs-assistant
# polkit-gnome: noctalia provides the agent (shell.polkit_agent)
polkit-gnome

# --- GNOME: the session is now niri only, login is handled by greetd with
# noctalia-greeter. The GNOME apps I use (nautilus, loupe, papers, ...)
# stay: they depend on the libraries, not on the shell.
gnome-shell
gdm

# NOTE: the NVIDIA and cuda packages are deliberately NOT listed here. On this
# laptop the GPU is not visible and they are 6 GB wasted, but a machine that
# actually has one needs them: the nvidia group installs them, and
# setup-machine.sh only offers it when it detects the GPU.
